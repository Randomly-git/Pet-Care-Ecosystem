public class CompleteWorkflowExample {

    @Autowired
    private PetService petService;
    @Autowired
    private RecordService recordService;

    public void completeWorkflow() {
        Long petId = 123L;

        // 1. 上传宠物头像
        MediaResponse avatar = petService.uploadPetAvatar(petId, "C:/pets/123/avatar.jpg");
        System.out.println("头像mediaId: " + avatar.getMediaId());

        // 2. 为活动记录456上传照片
        Long activityRecordId = 456L;
        MediaResponse activityPhoto = petService.uploadActivityFile(
            petId, activityRecordId, "D:/activities/456/photo1.jpg"
        );
        System.out.println("活动照片mediaId: " + activityPhoto.getMediaId());

        // 3. 为状态记录789上传健康报告
        Long statusRecordId = 789L;
        MediaResponse healthReport = petService.uploadStatusFile(
            petId, statusRecordId, "E:/status/789/report.pdf"
        );
        System.out.println("健康报告mediaId: " + healthReport.getMediaId());

        // 4. 删除活动记录及其所有文件
        recordService.deleteActivityRecord(petId, activityRecordId);
        System.out.println("活动记录及文件已删除");

        // 5. 获取状态记录的所有文件
        List<MediaResponse> statusFiles = recordService.getActivityFiles(statusRecordId);
        System.out.println("状态记录文件数量: " + statusFiles.size());
    }
}
