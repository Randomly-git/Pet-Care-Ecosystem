@RestController
public class ActivityController {

    @Autowired
    private PetService petService;

    /**
     * 为某个活动记录上传照片
     */
    @PostMapping("/activities/{activityRecordId}/photos")
    public ResponseEntity<?> uploadActivityPhoto(
            @PathVariable Long activityRecordId,
            @RequestParam Long petId,
            @RequestParam String filePath) {

        try {
            MediaResponse response = petService.uploadActivityFile(
                petId, activityRecordId, filePath
            );

            return ResponseEntity.ok(Map.of(
                "success", true,
                "mediaId", response.getMediaId(),
                "fileUrl", response.getFileUrl(),
                "message", "活动照片上传成功"
            ));
        } catch (Exception e) {
            return ResponseEntity.badRequest().body(Map.of(
                "success", false,
                "message", "上传失败: " + e.getMessage()
            ));
        }
    }
}
