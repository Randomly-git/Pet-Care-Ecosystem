@Service
@RequiredArgsConstructor
public class RecordService {

    private final MediaServiceFacade mediaServiceFacade;

    /**
     * 获取活动记录的所有文件
     */
    public List<MediaResponse> getActivityFiles(Long activityRecordId) {
        // 这里需要先知道宠物ID，或者修改MediaService支持按relatedId查询
        // 假设我们知道宠物ID是123L
        List<MediaResponse> allPetFiles = mediaServiceFacade.getPetFiles(123L);

        return allPetFiles.stream()
            .filter(file -> "ACTIVITY".equals(file.getRelatedType()))
            .filter(file -> activityRecordId.equals(file.getRelatedId()))
            .collect(Collectors.toList());
    }

    /**
     * 删除活动记录时，同时删除所有相关文件
     */
    public void deleteActivityRecord(Long petId, Long activityRecordId) {
        // 删除所有与该活动记录关联的文件
        mediaServiceFacade.deleteRelatedFiles("ACTIVITY", activityRecordId);

        // 然后删除活动记录本身
        // activityRepository.deleteById(activityRecordId);
    }

    /**
     * 删除状态记录时，同时删除所有相关文件
     */
    public void deleteStatusRecord(Long petId, Long statusRecordId) {
        // 删除所有与该状态记录关联的文件
        mediaServiceFacade.deleteRelatedFiles("STATUS", statusRecordId);

        // 然后删除状态记录本身
        // statusRepository.deleteById(statusRecordId);
    }
}
