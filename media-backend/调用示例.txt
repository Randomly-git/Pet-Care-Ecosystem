@Service
@RequiredArgsConstructor
public class PetService {

    private final MediaServiceFacade mediaServiceFacade;

    /**
     * 上传宠物头像 - 直接与宠物绑定
     */
    public MediaResponse uploadPetAvatar(Long petId, String avatarPath) {
        return mediaServiceFacade.uploadLocalFile(
            avatarPath,
            petId,                    // 宠物ID
            "PET_AVATAR",             // 宠物头像类型
            petId                     // 关联ID = 宠物ID（表示直接绑定到宠物）
        );
    }

    /**
     * 上传活动记录相关文件
     */
    public MediaResponse uploadActivityFile(Long petId, Long activityRecordId, String filePath) {
        return mediaServiceFacade.uploadLocalFile(
            filePath,
            petId,                    // 宠物ID
            "ACTIVITY",               // 活动记录类型
            activityRecordId          // 关联ID = 活动记录ID
        );
    }

    /**
     * 上传状态记录相关文件
     */
    public MediaResponse uploadStatusFile(Long petId, Long statusRecordId, String filePath) {
        return mediaServiceFacade.uploadLocalFile(
            filePath,
            petId,                    // 宠物ID
            "STATUS",                 // 状态记录类型
            statusRecordId            // 关联ID = 状态记录ID
        );
    }

    /**
     * 上传动态文件
     */
    public MediaResponse uploadMomentFile(Long petId, Long momentRecordId, String filePath) {
        return mediaServiceFacade.uploadLocalFile(
            filePath,
            petId,                    // 宠物ID
            "MOMENT",                 // 动态类型
            momentRecordId            // 关联ID = 动态记录ID
        );
    }
}
